CREATE DATABASE IF NOT EXISTS firma_db;

USE firma_db;

-- =========================================
-- TABELA: LimiteDeCredito
-- =========================================

CREATE TABLE LimiteDeCredito (
    id_limite INT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL
);

-- =========================================
-- TABELA: Endereco
-- =========================================

CREATE TABLE Endereco (
    id_endereco INT PRIMARY KEY,
    logradouro VARCHAR(150) NOT NULL,
    numero VARCHAR(20) NOT NULL,
    cep VARCHAR(10) NOT NULL,
    cidade VARCHAR(100) NOT NULL
);

-- =========================================
-- TABELA: Cliente
-- =========================================

CREATE TABLE Cliente (
    codigo INT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    sobreNome VARCHAR(100) NOT NULL,
    cpf VARCHAR(14) NOT NULL UNIQUE,
    telefone VARCHAR(20) NOT NULL,
    id_limite INT NOT NULL,
    id_endereco INT NOT NULL,

    CONSTRAINT fk_cliente_limite
        FOREIGN KEY (id_limite)
        REFERENCES LimiteDeCredito(id_limite),

    CONSTRAINT fk_cliente_endereco
        FOREIGN KEY (id_endereco)
        REFERENCES Endereco(id_endereco)
);

-- =========================================
-- TABELA: Produto
-- =========================================

CREATE TABLE Produto (
    codigo INT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao VARCHAR(255),
    preco DECIMAL(10,2) NOT NULL
);

-- =========================================
-- TABELA: Pedido
-- =========================================

CREATE TABLE Pedido (
    numero INT PRIMARY KEY,
    data_elaboracao DATE NOT NULL,
    id_cliente INT NOT NULL,

    CONSTRAINT fk_pedido_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES Cliente(codigo)
);

-- =========================================
-- TABELA: Pedido_Produto
-- =========================================

CREATE TABLE Pedido_Produto (
    id_pedido INT NOT NULL,
    id_produto INT NOT NULL,

    PRIMARY KEY (id_pedido, id_produto),

    CONSTRAINT fk_pp_pedido
        FOREIGN KEY (id_pedido)
        REFERENCES Pedido(numero),

    CONSTRAINT fk_pp_produto
        FOREIGN KEY (id_produto)
        REFERENCES Produto(codigo)
);
